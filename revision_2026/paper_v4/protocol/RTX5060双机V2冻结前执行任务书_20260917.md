```python
"""Read-only tree identity audit.

Compare every protocol's identity_files pins against the files actually present
in this checkout.  New file; it modifies nothing and is not pinned by any protocol.

Usage:
    python tools/verify_tree_identity.py --out analysis/<run>/tree_identity_audit.json
"""
from __future__ import annotations

import argparse
import hashlib
import json
import platform
import sys
from pathlib import Path


def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", type=Path, required=True)
    parser.add_argument("--root", type=Path, default=Path(__file__).resolve().parents[1])
    args = parser.parse_args()
    root = args.root.resolve()
    report = {"root": str(root), "host": platform.node(), "protocols": []}
    match = stale = no_pins = 0
    for path in sorted((root / "protocol").glob("*.json")):
        try:
            data = json.loads(path.read_text(encoding="utf-8"))
        except Exception as exc:                       # noqa: BLE001 - reported, not raised
            no_pins += 1
            report["protocols"].append({"protocol": path.name, "status": "UNPARSABLE", "error": str(exc)})
            continue
        items = data.get("identity_files")
        if not items:
            no_pins += 1
            report["protocols"].append({"protocol": path.name, "status": "NO_IDENTITY_FILES"})
            continue
        missing, mismatched = [], []
        for item in items:
            target = root / item["path"]
            if not target.is_file():
                missing.append(item["path"])
            elif sha256(target) != str(item["sha256"]).lower():
                mismatched.append(item["path"])
        status = "MATCH" if not missing and not mismatched else "STALE"
        match += status == "MATCH"
        stale += status == "STALE"
        report["protocols"].append({
            "protocol": path.name,
            "status": status,
            "identity_file_count": len(items),
            "missing": missing,
            "mismatched": mismatched,
            "protocol_sha256": sha256(path),
        })
    report["summary"] = {"protocols_total": len(report["protocols"]),
                         "match": match, "stale": stale, "no_identity_files": no_pins}
    args.out.parent.mkdir(parents=True, exist_ok=True)
    args.out.write_text(json.dumps(report, indent=2), encoding="utf-8")
    print(json.dumps(report["summary"], ensure_ascii=False))


if __name__ == "__main__":
    sys.exit(main())
```

3. 运行并把输出放进 `analysis/20260917_RTX5060_PRE_V2_<Mn>01/`：

**判据（唯一的拷贝完好判据）**：把本机的 `tree_identity_audit.json` 与 **5080 现场的同一脚本输出逐字段比对**。两者必须**完全一致**（含每份协议的 `status`、`missing`、`mismatched` 列表与 `protocol_sha256`）。

- 一致 → 拷贝无损，`STALE` 是**预期失效**（原因见第 1 节）；
- 不一致 → **停**。先比对不一致文件的大小与 SHA，判断是拷贝损坏还是两侧树本来就不同；**不要**为了让脚本变绿而改动任何被钉住的文件。
